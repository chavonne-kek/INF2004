cmake_minimum_required(VERSION 3.13)
set(PICO_BOARD pico_w)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(imu_sdk_probe C CXX ASM)
pico_sdk_init()
add_executable(imu_sdk_probe main.c lsm303d.c)
target_link_libraries(imu_sdk_probe pico_stdlib hardware_i2c)
pico_enable_stdio_usb(imu_sdk_probe 1)
pico_enable_stdio_uart(imu_sdk_probe 0)
pico_add_extra_outputs(imu_sdk_probe)
