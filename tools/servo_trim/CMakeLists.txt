cmake_minimum_required(VERSION 3.13)
set(PICO_BOARD pico_w)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(servo_trim C CXX ASM)
pico_sdk_init()
add_executable(servo_trim main.c)
target_link_libraries(servo_trim pico_stdlib hardware_pwm hardware_clocks)
pico_enable_stdio_usb(servo_trim 1)
pico_enable_stdio_uart(servo_trim 0)
pico_add_extra_outputs(servo_trim)
